function updatePerspectiveTelemetry() {
const dot = document.querySelector('.telemetry-dot');
const label = document.querySelector('.telemetry-label');
const container = document.getElementById('telemetryAiStatus');
if (!dot || !label) return;
+ if (activePerspectiveId === 'domain') {
+ dot.className = 'telemetry-dot local';
+ label.textContent = '⚡ Local (0 tokens)';
+ if (container) container.title = 'Gom nhóm theo hostname nội bộ — 0ms, không dùng AI';
+ } else {
+ dot.className = 'telemetry-dot ready';
+ label.textContent = openRouterApiKey ? 'OpenRouter (Jev)' : 'Smart Local';
+ if (container) container.title = 'Phân loại ngữ nghĩa bằng AI';
+ }
