function updatePerspectiveTelemetry() {
if (typeof document === 'undefined' || !document.querySelector) return;
const dot = document.querySelector('.telemetry-dot');
const aiLabel = document.querySelector('.telemetry-label');
const container = document.getElementById('telemetryAiStatus');
if (!dot || !aiLabel) return;
+ if (activePerspectiveId === 'domain') {
+ dot.className = 'telemetry-dot local';
+ aiLabel.textContent = 'Local rules';
+ if (container) container.title = 'Gom nhóm theo tên miền nội bộ — 0ms, không dùng AI';
+ } else if (aiAuthBlocked) {
+ dot.className = 'telemetry-dot auth-blocked';
+ aiLabel.textContent = 'Auth Blocked';
+ if (container) container.title = 'Lỗi xác thực OpenRouter (401/402/403) — Vui lòng kiểm tra API key';
+ } else {
+ dot.className = 'telemetry-dot ready';
+ aiLabel.textContent = openRouterApiKey ? 'OpenRouter (Jev)' : 'Smart Local';
+ if (container) container.title = 'Trạng thái phân loại ngữ nghĩa OpenRouter';
+ }
}
